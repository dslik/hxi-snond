Because valid UUIDs consist entirely of ASCII characters, using strlen to obtain the byte count is correct and does not conflict with the UTF-8 string handling requirement of Rule 1.2. ### Interactions | Parameter | Direction | Type | Description | | ------------ | --------- | ----- | ------------------------------------------------------------------ | | uuid_value | in | char* | Null-terminated string to validate | | return value | out | bool | true if the string is a well-formed RFC 4122 UUID; false otherwise | */ bool is_valid_uuid(const char* uuid_value) { uint32_t char_index = 0; REQUIRES(NULL != uuid_value, "NULL pointer passed for uuid_value"); if(C_UUID_STRLEN != strlen(uuid_value)) { return(false); } while(char_index < C_UUID_STRLEN) { if(C_UUID_HYPHEN_1 == char_index || C_UUID_HYPHEN_2 == char_index || C_UUID_HYPHEN_3 == char_index || C_UUID_HYPHEN_4 == char_index) { if('-' != uuid_value[char_index]) { return(false); } } else { if(0 == isxdigit((unsigned char) uuid_value[char_index])) { return(false); } } char_index = char_index + 1; } return(true); } /* ## Generate a random RFC 4122 version 4 UUID string Reads 16 bytes of cryptographically random data from /dev/urandom, applies the version (4) and variant (1) bit fields required by RFC 4122 section 4.4, and formats the result as a 36-character null-terminated UUID string of the form `xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx`. The caller must supply a buffer of at least 37 bytes. ### Interactions | Parameter | Direction | Type | Description | | ------------ | --------- | ------- | -------------------------------------------------------------------------- | | uuid_value | out | char* | Caller-allocated buffer of at least 37 bytes; receives the UUID string | | return value | out | int32_t | 0 on success; -1 if /dev/urandom cannot be opened or read | Other changes to system state include: - Opens and closes a file descriptor for /dev/urandom. - Reads 16 bytes from /dev/urandom. - Writes diagnostic messages to stderr on failure. */ int32_t generate_random_uuid(char* uuid_value) { uint8_t uuid_bytes[16] = {0}; int32_t urandom_fd = -1; int64_t read_size = 0; int32_t status = C_SUCCESS; REQUIRES(NULL != uuid_value, "NULL pointer passed for uuid_value"); urandom_fd = int32_from_int(open("/dev/urandom", O_RDONLY)); if(C_FAILURE == urandom_fd) { fprintf(stderr, "Unable to open /dev/urandom for UUID generation\n"); perror("/dev/urandom"); status = C_FAILURE; } else { read_size = read(urandom_fd, uuid_bytes, sizeof(uuid_bytes)); if(read_size != int64_from_ssize(sizeof(uuid_bytes))) { fprintf(stderr, "Unable to read from /dev/urandom for UUID generation\n"); perror("/dev/urandom"); status = C_FAILURE; } else { uuid_bytes[6] = (uuid_bytes[6] & 0x0F) | 0x40U; // Version 4 uuid_bytes[8] = (uuid_bytes[8] & 0x3F) | 0x80U; // Variant 1 snprintf(uuid_value, C_UUID_LEN, "%02x%02x%02x%02x-%02x%02x-%02x%02x-%02x%02x-%02x%02x%02x%02x%02x%02x", uuid_bytes[0], uuid_bytes[1], uuid_bytes[2], uuid_bytes[3], uuid_bytes[4], uuid_bytes[5], uuid_bytes[6], uuid_bytes[7], uuid_bytes[8], uuid_bytes[9], uuid_bytes[10], uuid_bytes[11], uuid_bytes[12], uuid_bytes[13], uuid_bytes[14], uuid_bytes[15]); } close(urandom_fd); } ENSURES(C_UUID_STRLEN == strlen(uuid_value), "Generated UUID has unexpected length %zu", strlen(uuid_value)); return(status); } //
